2.2：数据转换
2.2.1
浮点数转化为4个uchar 类型的数据
unsigned char* CXE_DoITDlg::DataAnla(double f,unsigned char kk[4])///浮点数转化为 4个char
{ 
 if(f<0)//f中的内容为负数
 {
  f*=(-1);//将F中的内容转换为正数
     int a=int(f);
   kk[0]=a/256+0x80;//将F中的内容转换为负数
   kk[1]=a%256;
   a=int((f-a)*10000);
   kk[2]=a/256;
   kk[3]=a%256;
 }
 else
 {
   int a=int(f);
   kk[0]=a/256;
   kk[1]=a%256;
   a=int((f-a+0.000001)*10000);
   kk[2]=a/256;
   kk[3]=a%256;
 } 
 return kk;
}
2.2.2
//数据转换函数
double CXE_DoITDlg::CalData(unsigned char kk0,unsigned char kk1,unsigned char kk2,unsigned char kk3)
{
 double d;
 if(kk0&0x80)
 {kk0-=0x80;
   d=(double)(kk0*256+kk1+(kk2*256+kk3)*0.0001);
      d*=(-1);
 }
    else
 {
  d=(double)(kk0*256+kk1+(kk2*256+kk3)*0.0001);
 }
 return d;
}

//例如:
发送10.001电压；
aa 01 0b 00 00 00 00 0a 00 0a a0

发送10.001位移；
aa 01 0b 01 00 00 00 0a 00 0a a1

2.2.3波形类型
路(通道)编号:0,1,2,3…（最大支持256路）
波类型: 'Z':正弦波,'F':方波,'S':三角波,'J':锯齿波

2.2.4 字符说明
 'O':开环,'C':闭环;    
'I':输入,'O':输出;    
'H':停止,'R':运行;    
'A':模拟,'D':数字;    
'Y':确认,'N'否定

2.3：字节序：字节内高位在前、低位在后
多字节通讯先发送高字节
注意：
2.4：通信方式：RS422
2.4.1：串口参数：波特率：115200 19200 38400 115200,停止位：1；数据位：8；奇偶校验位：无

2.5：帧间隔时间：最少50ms

2.6：校验方式：XOR-8校验
